async function loadApprovalsBatch() { const activeTab = currentApprovalTab; const cache = approvalsCache[activeTab]; if (isApprovalsLoading || !cache.hasMore) return; isApprovalsLoading = true; const online = await checkActualInternet(); if (!online) { showOfflineToast("Showing cached approvals."); const savedHtml = localStorage.getItem( "dakmitra_approvals_cache_" + activeTab ); const contentEl = document.getElementById("approvals-content"); const sentinelEl = document.getElementById("approvals-sentinel"); if (contentEl) { contentEl.innerHTML = savedHtml || `
📴 No internet connection and no cached ${activeTab} approvals saved yet.
`; } if (sentinelEl) { sentinelEl.innerHTML = `📴 Offline cached view`; } cache.hasMore = false; isApprovalsLoading = false; return; } const displayArea = document.getElementById("approvals-display-area"); let contentDiv = document.getElementById("approvals-content"); let sentinel = document.getElementById("approvals-sentinel"); if (!contentDiv) { displayArea.innerHTML = `
${getApprovalsSkeletonHTML(3)}
`; setupApprovalsObserver(); isApprovalsLoading = false; return setTimeout(loadApprovalsBatch, 0); } if (cache.page === 0) { if (!contentDiv.querySelector(".approval-skeleton-card")) { contentDiv.innerHTML = getApprovalsSkeletonHTML(3); } if (sentinel) sentinel.innerHTML = ""; } else if (sentinel) { sentinel.innerHTML = `
Fetching more requests...
`; } try { if (!currentUserId) { const { data: { session }, } = await supabaseClient.auth.getSession(); currentUserId = session?.user?.id || null; } const fromIndex = cache.page * APPROVALS_PAGE_SIZE; const toIndex = fromIndex + APPROVALS_PAGE_SIZE - 1; let query = supabaseClient .from("edit_requests") .select( "*, post_offices(Address, Office, Pin), request_approvals(approved_by, is_admin)", { count: "exact" } ) .order("created_at", { ascending: false }) .range(fromIndex, toIndex); if (activeTab === "pending") { query = query.eq("status", "PENDING"); } else { query = query.neq("status", "PENDING"); } const { data, error, count } = await query; if (error) throw error; // Ignore response if the user switched tabs while the query was in flight if (currentApprovalTab !== activeTab) { isApprovalsLoading = false; return; } if (!data || data.length === 0) { cache.hasMore = false; if (cache.page === 0) { const emptyMsg = activeTab === "pending" ? "No pending approvals! 🎉" : "No resolved requests yet."; const emptyHtml = `
${emptyMsg}
`; contentDiv.innerHTML = emptyHtml; // Cache the empty state so it doesn't re-fetch until something changes cache.html = emptyHtml; cache.page = 1; cache.initialHasMore = false; cache.loaded = true; } if (sentinel) sentinel.style.display = "none"; isApprovalsLoading = false; return; } // Check if we've reached the end of the records if ( data.length < APPROVALS_PAGE_SIZE || fromIndex + data.length >= count ) { cache.hasMore = false; if (sentinel) { sentinel.innerHTML = `End of requests`; } } else if (sentinel) { sentinel.innerHTML = ""; } // Gather all unique user IDs to fetch in one batch let userIdsToFetch = new Set(); data.forEach((req) => { if (req.requested_by) userIdsToFetch.add(req.requested_by); if (req.last_approved_by) userIdsToFetch.add(req.last_approved_by); if (req.request_approvals) { req.request_approvals.forEach((app) => { if (app.approved_by) userIdsToFetch.add(app.approved_by); }); } }); // Fetch profiles let profilesMap = {}; if (userIdsToFetch.size > 0) { const { data: profileData } = await supabaseClient .from("profiles") .select("id, name, employee_id") .in("id", Array.from(userIdsToFetch)); if (profileData) { profileData.forEach((p) => { profilesMap[p.id] = p; }); } } let html = ``; data.forEach((req) => { let changesHtml = ""; const formatApprovalField = ( oldVal, newVal, prefix = "", fallback = "N/A" ) => { const rawOld = oldVal === undefined || oldVal === null ? "" : String(oldVal).trim(); const rawNew = newVal === undefined || newVal === null ? "" : String(newVal).trim(); const sOld = rawOld === "" ? fallback : rawOld; const sNew = rawNew === "" ? fallback : rawNew; if (sOld !== sNew) { return `
${prefix} ${escapeHTML( sOld )} ➡️ ${escapeHTML(sNew)}
`; } return `
${prefix} ${escapeHTML( sNew )}
`; }; if (req.action_type === "ADD") { const badge = `➕ ADD`; const prop = req.proposed_data || {}; changesHtml = ` ${badge}
${escapeHTML( prop.Address || "Unknown" )}
🏢 ${escapeHTML( prop.Office || "Unknown" )}
📍 ${escapeHTML( prop.Pin || "N/A" )}
`; } else if (req.action_type === "EDIT") { const badge = `✏️ EDIT`; const oldD = req.original_data || req.post_offices || {}; const newD = req.proposed_data || {}; changesHtml = badge + formatApprovalField( oldD.Address || oldD.address, newD.Address || newD.address, "📮", "Unknown Address" ) + formatApprovalField( oldD.Office || oldD.office, newD.Office || newD.office, "🏢", "Unknown Office" ) + formatApprovalField( oldD.Pin || oldD.pin, newD.Pin || newD.pin, "📍", "N/A" ); } else if (req.action_type === "DELETE") { const badge = `🗑️ DELETE`; const oldD = req.post_offices || {}; changesHtml = ` ${badge}
${escapeHTML( oldD.Address || "Unknown" )}
🏢 ${escapeHTML( oldD.Office || "Unknown" )}
📍 ${escapeHTML( oldD.Pin || "N/A" )}
`; } const voteCount = req.request_approvals ? req.request_approvals.length : 0; const dateStr = new Date(req.created_at).toLocaleString("en-IN", { dateStyle: "medium", timeStyle: "short", }); const reqProfile = profilesMap[req.requested_by]; const requesterName = reqProfile ? reqProfile.name : "Unknown User"; const requesterId = reqProfile && reqProfile.employee_id ? reqProfile.employee_id : "No ID"; let approverDetailsHtml = ""; if ( req.status === "APPROVED" && req.request_approvals && req.request_approvals.length > 0 ) { const lastApproval = req.request_approvals[req.request_approvals.length - 1]; const approverProfile = profilesMap[lastApproval.approved_by]; if (approverProfile) { if (lastApproval.is_admin) { approverDetailsHtml = `
✅ Admin Approved (Approved by: ${escapeHTML( approverProfile.name )} - ${escapeHTML(approverProfile.employee_id)})
`; } else { approverDetailsHtml = `
👥 Community Approved (Final vote by: ${escapeHTML( approverProfile.name )} - ${escapeHTML(approverProfile.employee_id)})
`; } } } let actionBarHtml = ""; if (req.status === "PENDING") { const isOwnRequest = currentUserId && String(req.requested_by) === String(currentUserId); const hasAlreadyVoted = currentUserId && req.request_approvals && req.request_approvals.some( (app) => String(app.approved_by) === String(currentUserId) ); actionBarHtml = `
${ isOwnRequest ? `⏳ Your request (awaiting votes) ` : "" } ${ isCurrentUserAdmin && !isOwnRequest ? `` : "" } ${ !isOwnRequest && !hasAlreadyVoted ? `` : "" } ${ !isOwnRequest && hasAlreadyVoted ? `✓ You Voted` : "" }
`; } else if (req.status === "APPROVED") { actionBarHtml = `
✅ Request Approved & Merged
`; } else if (req.status === "REJECTED") { let rejecterText = ""; if (req.last_approved_by && profilesMap[req.last_approved_by]) { const rej = profilesMap[req.last_approved_by]; rejecterText = ` (Rejected by: ${escapeHTML( rej.name )} - ${escapeHTML(rej.employee_id)})`; } actionBarHtml = `
❌ Request Rejected${rejecterText}
`; } html += `
${changesHtml}
👤 Requested by: ${escapeHTML( requesterName )} (${escapeHTML(requesterId)})
📅 ${dateStr}
${ req.status === "PENDING" ? `
🗳️ Community Votes: ${voteCount} / 5
` : "" } ${approverDetailsHtml}
${actionBarHtml}
`; }); if (cache.page === 0) { contentDiv.classList.remove("approvals-smooth-in"); contentDiv.innerHTML = html; void contentDiv.offsetWidth; // Restarts animation cleanly contentDiv.classList.add("approvals-smooth-in"); } else { contentDiv.insertAdjacentHTML("beforeend", html); } // Save ONLY the first 5 cards (page 0) in memory & localStorage cache if (cache.page === 0) { cache.html = html; cache.loaded = true; cache.initialHasMore = cache.hasMore; cache.page = 1; localStorage.setItem("dakmitra_approvals_cache_" + activeTab, html); } else { cache.page++; } isApprovalsLoading = false; } catch (err) { isApprovalsLoading = false; if (contentDiv) contentDiv.innerHTML = `
Error loading requests: ${err.message}
`; } }
