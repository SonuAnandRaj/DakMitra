let xlsxLoadPromise = null;
function ensureXLSXLoaded() {
  if (window.XLSX) return Promise.resolve(window.XLSX);
  if (!xlsxLoadPromise) {
    xlsxLoadPromise = new Promise((resolve, reject) => {
      const s = document.createElement("script");
      s.src = "https://cdn.jsdelivr.net/npm/xlsx/dist/xlsx.full.min.js";
      s.onload = () => resolve(window.XLSX);
      s.onerror = () => {
        xlsxLoadPromise = null;
        reject(new Error("Failed to load Excel library."));
      };
      document.head.appendChild(s);
    });
  }
  return xlsxLoadPromise;
}

// --- SAFE STORAGE & ESCAPING ---
function getSafeStorage(key) {
  try {
    return JSON.parse(localStorage.getItem(key)) || [];
  } catch (e) {
    console.warn(`Resetting corrupted storage for ${key}`);
    return [];
  }
}

function escapeHTML(str) {
  if (str === null || str === undefined) return "";
  return String(str)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}

function escapeForOnclick(str) {
  if (!str) return "";
  return String(str)
    .replace(/\\/g, "\\\\")
    .replace(/'/g, "\\'")
    .replace(/"/g, "&quot;");
}

// --- BACKGROUND "LIE-FI" DETECTOR (Use ONLY for silent reconnect / wake-up) ---
async function checkActualInternet(timeoutMs = 2500) {
  if (!navigator.onLine) return false;
  try {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), timeoutMs);

    await fetch("https://clients3.google.com/generate_204", {
      method: "HEAD",
      mode: "no-cors",
      cache: "no-store",
      signal: controller.signal,
    });

    clearTimeout(timeoutId);
    return true;
  } catch (e) {
    return false;
  }
}

function isNetworkError(err) {
  if (!navigator.onLine) return true;
  const msg = String(err?.message || err || "").toLowerCase();
  return (
    msg.includes("failed to fetch") ||
    msg.includes("networkerror") ||
    msg.includes("network request failed") ||
    msg.includes("load failed") ||
    msg.includes("err_internet") ||
    msg.includes("err_name_not_resolved") ||
    msg.includes("aborted")
  );
}

function showOfflineToast(
  extraMsg = "Using cached version.",
  isActiveAction = false,
) {
  if (!isActiveAction && hasShownPassiveOfflineToast) return;

  const now = Date.now();
  if (now - lastOfflineToastTime < 2000) return;
  lastOfflineToastTime = now;

  if (!isActiveAction) hasShownPassiveOfflineToast = true;
  showToast(`ðŸ“´ No internet connection. ${extraMsg}`);
}

// For DB writes/actions: only block if the OS reports offline; let slow networks connect to Supabase
async function ensureOnline(
  offlineMsg = "Using cached version.",
  isActiveAction = true,
) {
  if (!navigator.onLine) {
    showOfflineToast(offlineMsg, isActiveAction);
    return false;
  }
  hasShownPassiveOfflineToast = false;
  return true;
}

function showToast(message, type = null) {
  const toastEl = document.getElementById("toast");
  if (!toastEl) return;
  clearTimeout(toastTimer);

  const resolvedType = type || detectToastType(message);
  toastEl.className = `toast toast-${resolvedType}`;
  toastEl.textContent = message;

  requestAnimationFrame(() => toastEl.classList.add("show"));
  toastTimer = setTimeout(() => toastEl.classList.remove("show"), 3000);

  if (navigator.vibrate) {
    navigator.vibrate(resolvedType === "error" ? [40, 40, 40] : 50);
  }
}

// --- UNIFIED BUTTON LOADING / SUCCESS STATE HELPER ---
function setButtonState(btn, state, label = "") {
  if (!btn) return;
  if (!btn.dataset.originalHtml) {
    btn.dataset.originalHtml = btn.innerHTML;
  }

  btn.classList.remove("is-loading", "is-success");

  if (state === "loading") {
    btn.disabled = true;
    btn.classList.add("is-loading");
    btn.innerHTML = `<div class="btn-spinner"></div><span>${escapeHTML(
      label || "Working...",
    )}</span>`;
  } else if (state === "success") {
    btn.disabled = true;
    btn.classList.add("is-success");
    btn.innerHTML = `
      <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3" stroke-linecap="round" stroke-linejoin="round">
        <polyline class="checkmark-path" points="20 6 9 17 4 12"></polyline>
      </svg>
      <span>${escapeHTML(label || "Done!")}</span>
    `;
  } else {
    btn.disabled = false;
    btn.innerHTML = label || btn.dataset.originalHtml;
  }
}

// --- UNIFIED POSTAL PHONETIC KEY (EXACT 1:1 PORT OF SUPABASE public.postal_phonetic_key) ---
const _phoneticKeyCache = new Map();

function postalPhoneticKey(addr) {
  if (addr === null || addr === undefined) return "";
  const input = String(addr);
  if (_phoneticKeyCache.has(input)) return _phoneticKeyCache.get(input);

  // 1. Lowercase, remove dots, and strip standalone BO/SO/HO/PO/Village tags
  let s = input.replace(/\./g, "").toLowerCase();
  s = s.replace(/\b(bo|so|ho|po|village|vill|gram)\b/gi, "");

  // 2. Normalize common suffix spellings (Kalan/Kalna -> Kala, Gawn/Gawan/Ganw -> Gaon)
  s = s.replace(/\b(kalan|kalna)\b/gi, "kala");
  s = s.replace(/(gawn|gawan|ganw|gaon)\b/gi, "gaon");

  // 3. Remove all spaces, hyphens, and non-alphanumeric characters
  s = s.replace(/[^a-z0-9]/g, "");

  // 4. Normalize Hindi consonant transliterations
  s = s
    .replace(/chh/g, "c")
    .replace(/ch/g, "c")
    .replace(/kh/g, "k")
    .replace(/gh/g, "g")
    .replace(/th/g, "t")
    .replace(/dh/g, "d")
    .replace(/sh/g, "s")
    .replace(/w/g, "v")
    .replace(/d/g, "r");

  // 5. Normalize vowel variations
  s = s
    .replace(/ee/g, "i")
    .replace(/oo/g, "u")
    .replace(/aa/g, "a")
    .replace(/iya/g, "ia")
    .replace(/ue/g, "i")
    .replace(/ui/g, "i")
    .replace(/ai/g, "e")
    .replace(/au/g, "o");

  // 6. Normalize trailing nasal 'n' or silent 'h' after 'a'
  s = s.replace(/a[nh]$/, "a");

  // 7. Remove internal schwa 'a' vowels while preserving first and last character
  if (s.length > 2) {
    s = s[0] + s.slice(1, -1).replace(/a/g, "") + s[s.length - 1];
  }

  // 8. Collapse consecutive duplicate letters (e.g. Bichchhi -> Bici)
  s = s.replace(/(.)\1+/g, "$1");

  if (_phoneticKeyCache.size > 5000) _phoneticKeyCache.clear();
  _phoneticKeyCache.set(input, s);
  return s;
}

// --- SHARED OFFICE TYPE & EFFECTIVE PRIORITY HELPERS ---
function detectOfficeType(address) {
  if (!address) return null;
  const clean = String(address).replace(/\./g, "");
  if (/\b(HO|GPO)\b/i.test(clean)) return "HO";
  if (/\bSO\b/i.test(clean)) return "SO";
  if (/\b(BO|EDBO)\b/i.test(clean)) return "BO";
  return null;
}

function stripPostalSuffixes(text) {
  if (!text) return "";
  return String(text)
    .replace(/\./g, "")
    .toLowerCase()
    .replace(/\b(edbo|gpo|bo|so|ho|po|village|vill|gram)\b/gi, " ")
    .replace(/[^a-z0-9]+/g, " ")
    .trim();
}

/**
 * Effective Priority Tiers (higher = ranks first):
 * 4 = HO / SO (Automatic top)
 * 3 = BO + manually marked Frequent (breaks ties between same-name BOs)
 * 2 = BO (Automatic above regular villages)
 * 1 = Non-BO village manually marked Frequent / Important
 * 0 = Normal scattered village
 */
function getEffectivePriority(row) {
  if (!row) return 0;
  const officeType = detectOfficeType(row.Address);
  const isPriority = Number(row.priority) >= 1;
  if (officeType === "HO" || officeType === "SO") return 4;
  if (officeType === "BO") return isPriority ? 3 : 2;
  return isPriority ? 1 : 0;
}

// --- SMOOTH SPLASH CONTROLLER ---
function showSplashScreen() {
  const splash = document.getElementById("splash-screen");
  const content = document.getElementById("splash-content");
  if (!splash) return;
  if (content) {
    const h = window.visualViewport
      ? window.visualViewport.height
      : window.innerHeight;
    if (h > 0) content.style.top = Math.round(h / 2) + "px";
  }
  splash.classList.remove("hidden-splash");
}

function hideSplashScreen() {
  const splash = document.getElementById("splash-screen");
  if (!splash || splash.classList.contains("hidden-splash")) return;

  const elapsed = performance.now() - SPLASH_START_TIME;
  const remaining = Math.max(0, SPLASH_MIN_DURATION_MS - elapsed);

  setTimeout(() => {
    requestAnimationFrame(() => {
      splash.classList.add("hidden-splash");
    });
  }, remaining);
}

// --- THEME LOGIC (Exact System Bar & Notch Color Match) ---
function applyTheme(theme) {
  let isDark = false;
  if (theme === "system") {
    isDark = window.matchMedia("(prefers-color-scheme: dark)").matches;
    document.documentElement.setAttribute(
      "data-theme",
      isDark ? "dark" : "light",
    );
  } else {
    isDark = theme === "dark";
    document.documentElement.setAttribute("data-theme", theme);
  }
  localStorage.setItem("app-theme", theme);

  const bgColor = isDark ? "#090e17" : "#f8fafc";
  document.documentElement.style.colorScheme = isDark ? "dark" : "light";
  document.documentElement.style.backgroundColor = bgColor;
  const metaTheme = document.getElementById("meta-theme-color");
  if (metaTheme) {
    metaTheme.setAttribute("content", bgColor);
  }
}

function toggleTheme() {
  const current = document.documentElement.getAttribute("data-theme");
  const newTheme = current === "dark" ? "light" : "dark";
  applyTheme(newTheme);
}

    if (typeof switchApprovalTab === "function") {
      switchApprovalTab(currentApprovalTab);
    }
  } else if (viewId === "directory") {
    directoryHeroController.setCollapsed(false, true);
    if (typeof renderDirectory === "function") renderDirectory();
  } else if (viewId === "members") {
    membersHeroController.setCollapsed(false, true);
    if (typeof renderMembersDirectory === "function") renderMembersDirectory();
  } else if (viewId === "feedback") {
    feedbackHeroController.setCollapsed(false, true);
    if (typeof renderFeedbackView === "function") renderFeedbackView();
  } else if (viewId === "learn") {
    if (typeof renderLearnView === "function") renderLearnView();
  }

  closeSidebar();
}

// --- PWA ANDROID / BROWSER BACK BUTTON HANDLER ---
(function initPwaBackNavigation() {
  let isHandlingPopstate = false;
  let isRestoringFromPopstate = false;
  let rootGuardPushed = false;

  function isRootCleanState() {
    const hasOpenModal = !!document.querySelector(".modal-overlay.show");
    const hasOpenSidebar = !!document
      .getElementById("sidebar")
      ?.classList.contains("open");
    const hasOpenPopover = !!document
      .getElementById("directory-floating-bar")
      ?.classList.contains("open");
    const searchActive = !!document
      .getElementById("view-search")
      ?.classList.contains("active");

    return !hasOpenModal && !hasOpenSidebar && !hasOpenPopover && searchActive;
  }

  function pushNavStep(tag, viewId = null) {
    if (isHandlingPopstate || isRestoringFromPopstate) return;
    try {
      const stateObj = { dakmitraNav: tag, ts: Date.now() };
      if (viewId) stateObj.viewId = viewId;
      history.pushState(stateObj, "");
      rootGuardPushed = true;
    } catch (_) {}
  }

  window.pushPwaNavStep = pushNavStep;

  function ensureBaseHistoryState() {
    try {
      if (!history.state || !history.state.dakmitraBase) {
        history.replaceState({ dakmitraBase: true }, "");
      }
      if (!rootGuardPushed) {
        history.pushState({ dakmitraNav: "root", viewId: "search" }, "");
        rootGuardPushed = true;
      }
    } catch (_) {}
  }

  window.addEventListener("load", ensureBaseHistoryState);
  setTimeout(ensureBaseHistoryState, 0);

  const origToggleSidebar = window.toggleSidebar;
  if (typeof origToggleSidebar === "function") {
    window.toggleSidebar = function () {
      const wasOpen = document
        .getElementById("sidebar")
        ?.classList.contains("open");
      origToggleSidebar.apply(this, arguments);
      const isNowOpen = document
        .getElementById("sidebar")
        ?.classList.contains("open");
      if (!wasOpen && isNowOpen) {
        pushNavStep("sidebar");
      }
    };
  }

  const origSwitchView = window.switchDashboardView;
  if (typeof origSwitchView === "function") {
    window.switchDashboardView = function (viewId) {
      origSwitchView.apply(this, arguments);
      if (viewId) {
        pushNavStep("view:" + viewId, viewId);
      }
    };
  }

  const modalObserver = new MutationObserver((mutations) => {
    if (isHandlingPopstate) return;
    for (const m of mutations) {
      if (
        m.type === "attributes" &&
        m.attributeName === "class" &&
        m.target.classList?.contains("modal-overlay") &&
        m.target.classList.contains("show")
      ) {
        pushNavStep("modal:" + (m.target.id || "dialog"));
        break;
      }
    }
  });

  document.querySelectorAll(".modal-overlay").forEach((modal) => {
    modalObserver.observe(modal, { attributes: true });
  });

  window.addEventListener("popstate", (event) => {
    isHandlingPopstate = true;

    try {
      const openModals = Array.from(
        document.querySelectorAll(".modal-overlay.show"),
      );
      if (openModals.length > 0) {
        const topModal = openModals[openModals.length - 1];
        if (
          topModal.id === "edit-record-modal-overlay" &&
          window.closeEditRecordModal
        ) {
          closeEditRecordModal();
        } else if (
          topModal.id === "quick-add-modal-overlay" &&
          window.closeQuickAddModal
        ) {
          closeQuickAddModal();
        } else if (
          topModal.id === "quick-delete-modal-overlay" &&
          window.closeQuickDeleteModal
        ) {
          closeQuickDeleteModal();
        } else if (
          topModal.id === "quick-priority-modal-overlay" &&
          window.closeQuickPriorityModal
        ) {
          closeQuickPriorityModal();
        } else if (
          topModal.id === "feedback-admin-modal-overlay" &&
          window.closeFeedbackAdminModal
        ) {
          closeFeedbackAdminModal();
        } else if (
          topModal.id === "office-details-modal-overlay" &&
          window.closeOfficeDetailsModal
        ) {
          closeOfficeDetailsModal();
        } else if (
          topModal.id === "history-modal-overlay" &&
          window.closeHistoryModal
        ) {
          closeHistoryModal();
        } else if (
          topModal.id === "user-profile-modal-overlay" &&
          window.closeUserProfileModal
        ) {
          closeUserProfileModal();
        } else if (
          topModal.id === "crop-modal-overlay" &&
          window.closeCropModal
        ) {
          closeCropModal();
        } else if (
          topModal.id === "avatar-lightbox-overlay" &&
          window.closeProfilePhotoModal
        ) {
          closeProfilePhotoModal();
        } else if (
          topModal.id === "confirm-modal-overlay" &&
          window.closeConfirmModal
        ) {
          closeConfirmModal();
        } else if (
          topModal.id === "alert-modal-overlay" &&
          window.closeAlertModal
        ) {
          closeAlertModal();
        } else {
          topModal.classList.remove("show");
        }
        return;
      }

      const popover = document.getElementById("directory-floating-bar");
      if (popover && popover.classList.contains("open")) {
        if (typeof closeAllDirectoryActionTrays === "function") {
          closeAllDirectoryActionTrays();
        } else {
          popover.classList.remove("open");
        }
        return;
      }

      const sidebar = document.getElementById("sidebar");
      if (sidebar && sidebar.classList.contains("open")) {
        if (typeof closeSidebar === "function") {
          closeSidebar();
        } else {
          toggleSidebar();
        }
        return;
      }

      if (
        typeof handleLearnBackNavigation === "function" &&
        handleLearnBackNavigation()
      ) {
        return;
      }

      const targetView =
        event.state && typeof event.state.viewId === "string"
          ? event.state.viewId
          : "search";

      const currentViewId =
        document
          .querySelector(".view-section.active")
          ?.id?.replace(/^view-/, "") || "search";

      if (
        currentViewId !== targetView &&
        typeof origSwitchView === "function"
      ) {
        isRestoringFromPopstate = true;
        try {
          origSwitchView.call(window, targetView);
        } finally {
          setTimeout(() => {
            isRestoringFromPopstate = false;
          }, 0);
        }
        return;
      }

      if (isRootCleanState()) {
        showToast("Press Back again to exit DakMitra");
      }
    } finally {
      setTimeout(() => {
        isHandlingPopstate = false;
      }, 60);
    }
  });
})();

// --- APP-WIDE SMART KEYBOARD MANAGER ---
(function initGlobalKeyboardHandler() {
  const root = document.documentElement;
  const vv = window.visualViewport;
  let baselineHeight = Math.max(window.innerHeight, vv ? vv.height : 0);
  let lastWidth = vv ? vv.width : window.innerWidth;
  let blurTimer = null;

  function isEditableElement(el) {
    if (!el) return false;
    if (el.isContentEditable) return true;
    const tag = el.tagName;
    if (tag === "TEXTAREA") return !el.readOnly && !el.disabled;
    if (tag === "INPUT") {
      const nonText = [
        "checkbox",
        "radio",
        "file",
        "range",
        "button",
        "submit",
        "reset",
        "color",
        "hidden",
      ];
      return (
        !nonText.includes((el.type || "text").toLowerCase()) &&
        !el.readOnly &&
        !el.disabled
      );
    }
    return false;
  }

  function ensureFocusedInputVisible(el) {
    if (!isEditableElement(el)) return;
    if (!document.body.classList.contains("keyboard-open")) return;

    const vvTop = vv ? vv.offsetTop : 0;
    const vvHeight = vv ? vv.height : window.innerHeight;
    const rect = el.getBoundingClientRect();

    const insideModal = el.closest(".modal-overlay.show");
    const scrollParent =
      el.closest(
        ".feedback-submit-panel, .staff-detail-list, .results-container-scroll, #view-upload, #view-profile, #view-settings, #view-learn",
      ) ||
      insideModal ||
      null;

    if (scrollParent) {
      const parentRect = scrollParent.getBoundingClientRect();
      const safeTop = Math.max(
        insideModal ? vvTop + 20 : vvTop + 76,
        parentRect.top + 8,
      );
      const safeBottom = Math.min(
        vvTop + vvHeight - 20,
        parentRect.bottom - 12,
      );

      if (rect.top >= safeTop && rect.bottom <= safeBottom) return;

      const targetTop = rect.top - parentRect.top + scrollParent.scrollTop - 16;
      scrollParent.scrollTo({
        top: Math.max(0, targetTop),
        behavior: "smooth",
      });
    } else {
      const safeTop = vvTop + 76;
      const safeBottom = vvTop + vvHeight - 72;
      if (rect.top >= safeTop && rect.bottom <= safeBottom) return;

      const desiredTop = vvTop + Math.min(110, vvHeight * 0.25);
      window.scrollBy({
        top: rect.top - desiredTop,
        behavior: "smooth",
      });
    }
  }

  function syncKeyboardState() {
    const vvHeight = vv ? vv.height : window.innerHeight;
    const vvWidth = vv ? vv.width : window.innerWidth;
    const vvTop = vv ? vv.offsetTop : 0;

    if (Math.abs(vvWidth - lastWidth) > 50) {
      lastWidth = vvWidth;
      baselineHeight = Math.max(window.innerHeight, vvHeight);
    }

    const activeEl = document.activeElement;
    const activeIsEditable = isEditableElement(activeEl);

    if (!activeIsEditable && vvHeight > baselineHeight) {
      baselineHeight = vvHeight;
    }

    const kbHeight = Math.max(0, Math.round(baselineHeight - vvHeight));
    const fixedOffset = Math.max(
      0,
      Math.round(window.innerHeight - vvHeight - vvTop),
    );
    const isKeyboardOpen = activeIsEditable && kbHeight > 110;

    root.style.setProperty("--vvh", `${Math.round(vvHeight)}px`);
    root.style.setProperty("--vv-top", `${Math.round(vvTop)}px`);
    root.style.setProperty("--kb-height", `${isKeyboardOpen ? kbHeight : 0}px`);
    root.style.setProperty(
      "--kb-fixed-offset",
      `${isKeyboardOpen ? fixedOffset : 0}px`,
    );

    document.body.classList.toggle("keyboard-open", isKeyboardOpen);

    // Prevent Chrome from shifting the outer page up above the keyboard on dashboard views
    const isDashboardVisible = !document
      .getElementById("dashboard-view")
      ?.classList.contains("hidden");
    if (isDashboardVisible && window.scrollY > 0) {
      window.scrollTo(0, 0);
    }

    if (isKeyboardOpen) {
      requestAnimationFrame(() => ensureFocusedInputVisible(activeEl));
    }
  }

  if (vv) {
    vv.addEventListener("resize", syncKeyboardState);
    vv.addEventListener("scroll", syncKeyboardState);
  }
  window.addEventListener("resize", syncKeyboardState);

  document.addEventListener("focusin", (e) => {
    clearTimeout(blurTimer);
    if (!isEditableElement(e.target)) return;
    syncKeyboardState();
    setTimeout(() => {
      syncKeyboardState();
      ensureFocusedInputVisible(e.target);
    }, 180);
    setTimeout(() => ensureFocusedInputVisible(e.target), 360);
  });

  document.addEventListener("focusout", () => {
    clearTimeout(blurTimer);
    blurTimer = setTimeout(syncKeyboardState, 120);
  });
})();

// --- GLOBAL ACTIVE-WINDOW SCROLL LOCK (ALL MODALS & SIDEBAR) ---
(function initGlobalOverlayScrollLock() {
  let touchStartX = 0;
  let touchStartY = 0;

  function getTopActiveOverlay() {
    const active = Array.from(
      document.querySelectorAll(".modal-overlay.show, .sidebar.open"),
    );
    if (active.length === 0) return null;

    return active.sort((a, b) => {
      const za = parseInt(getComputedStyle(a).zIndex, 10) || 0;
      const zb = parseInt(getComputedStyle(b).zIndex, 10) || 0;
      return za - zb;
    })[active.length - 1];
  }

  function syncScrollLock() {
    const hasActiveOverlay = !!getTopActiveOverlay();
    document.documentElement.classList.toggle(
      "dm-scroll-locked",
      hasActiveOverlay,
    );
    document.body.classList.toggle("dm-scroll-locked", hasActiveOverlay);
  }

  function findScrollableAncestor(target, boundaryEl) {
    let el = target instanceof Element ? target : target?.parentElement;
    while (el) {
      const style = getComputedStyle(el);
      const canScrollY =
        (style.overflowY === "auto" || style.overflowY === "scroll") &&
        el.scrollHeight > el.clientHeight + 1;

      if (canScrollY) return el;
      if (el === boundaryEl) break;
      el = el.parentElement;
    }
    return null;
  }

  function findHorizontalScrollableAncestor(target, boundaryEl) {
    let el = target instanceof Element ? target : target?.parentElement;
    while (el) {
      const style = getComputedStyle(el);
      const canScrollX =
        (style.overflowX === "auto" || style.overflowX === "scroll") &&
        el.scrollWidth > el.clientWidth + 1;

      if (canScrollX) return el;
      if (el === boundaryEl) break;
      el = el.parentElement;
    }
    return null;
  }

  const observer = new MutationObserver(syncScrollLock);
  document
    .querySelectorAll(".modal-overlay, .sidebar, .sidebar-overlay")
    .forEach((el) => {
      observer.observe(el, {
        attributes: true,
        attributeFilter: ["class"],
      });
    });
  syncScrollLock();

  document.addEventListener(
    "touchstart",
    (e) => {
      if (e.touches.length === 1) {
        touchStartX = e.touches[0].clientX;
        touchStartY = e.touches[0].clientY;
      }
    },
    { passive: true },
  );

  document.addEventListener(
    "touchmove",
    (e) => {
      const activeOverlay = getTopActiveOverlay();
      if (!activeOverlay) return;

      if (e.target.closest("#crop-stage, #lightbox-stage")) return;
      if (e.target.tagName === "INPUT" && e.target.type === "range") return;

      if (!activeOverlay.contains(e.target)) {
        e.preventDefault();
        return;
      }

      if (!e.touches || e.touches.length === 0) return;
      const deltaX = e.touches[0].clientX - touchStartX;
      const deltaY = e.touches[0].clientY - touchStartY;

      if (
        Math.abs(deltaX) > Math.abs(deltaY) &&
        findHorizontalScrollableAncestor(e.target, activeOverlay)
      ) {
        return;
      }

      if (Math.abs(deltaY) < 8) return; // Allow normal taps without cancelling

      const scrollable = findScrollableAncestor(e.target, activeOverlay);
      if (!scrollable) {
        e.preventDefault();
        return;
      }

      const atTop = scrollable.scrollTop <= 0 && deltaY > 0;
      const atBottom =
        scrollable.scrollTop + scrollable.clientHeight >=
          scrollable.scrollHeight - 1 && deltaY < 0;

      if (atTop || atBottom) {
        e.preventDefault();
      }
    },
    { passive: false },
  );

  document.addEventListener(
    "wheel",
    (e) => {
      const activeOverlay = getTopActiveOverlay();
      if (!activeOverlay) return;

      if (!activeOverlay.contains(e.target)) {
        e.preventDefault();
        return;
      }

      if (
        Math.abs(e.deltaX) > Math.abs(e.deltaY) &&
        findHorizontalScrollableAncestor(e.target, activeOverlay)
      ) {
        return;
      }

      const scrollable = findScrollableAncestor(e.target, activeOverlay);
      if (!scrollable) {
        e.preventDefault();
        return;
      }

      const atTop = scrollable.scrollTop <= 0 && e.deltaY < 0;
      const atBottom =
        scrollable.scrollTop + scrollable.clientHeight >=
          scrollable.scrollHeight - 1 && e.deltaY > 0;

      if (atTop || atBottom) {
        e.preventDefault();
      }
    },
    { passive: false },
  );
})();

// --- ULTRA-SMOOTH COLLAPSIBLE HERO CONTROLLER (Search, Approvals, Directory, Feedback) ---
function createCollapsibleHero(
  scrollElIds,
  wrapElId,
  onCollapse = null,
  options = {},
) {
  const ids = Array.isArray(scrollElIds) ? scrollElIds : [scrollElIds];
  const scrollEls = ids
    .map((id) => document.getElementById(id))
    .filter(Boolean);
  const wrapEl = document.getElementById(wrapElId);

  const collapseAccum = options.collapseAccum ?? 240;
  const minScrollTop = options.minScrollTop ?? 180;
  const expandAccum = options.expandAccum ?? 40;

  let isCollapsed = false;
  let isTransitioning = false;
  let transitionTimer = null;
  let scrollAccum = 0;
  let activeScrollEl = scrollEls[0] || null;
  const lastScrollMap = new WeakMap();
  scrollEls.forEach((el) => lastScrollMap.set(el, 0));

  function syncAllScrollTops() {
    scrollEls.forEach((el) => {
      lastScrollMap.set(el, Math.max(0, el.scrollTop));
    });
  }

  function finishTransition() {
    clearTimeout(transitionTimer);
    if (wrapEl) wrapEl.classList.remove("is-animating");
    isTransitioning = false;
    scrollAccum = 0;

    if (activeScrollEl && activeScrollEl.offsetParent !== null) {
      lastScrollMap.set(activeScrollEl, Math.max(0, activeScrollEl.scrollTop));
      const remainingScroll =
        activeScrollEl.scrollHeight - activeScrollEl.clientHeight;
      if (isCollapsed && remainingScroll <= 20) {
        setCollapsed(false);
      }
    }
  }

  function setCollapsed(shouldCollapse, instant = false) {
    if (!wrapEl) return;

    if (isCollapsed === shouldCollapse) {
      if (instant) {
        scrollAccum = 0;
        syncAllScrollTops();
      }
      return;
    }

    isCollapsed = shouldCollapse;
    clearTimeout(transitionTimer);

    if (instant) {
      wrapEl.classList.remove("is-animating");
      wrapEl.classList.toggle("collapsed", shouldCollapse);
      isTransitioning = false;
      scrollAccum = 0;
      syncAllScrollTops();
      return;
    }

    isTransitioning = true;
    wrapEl.classList.add("is-animating");

    if (shouldCollapse) {
      if (typeof onCollapse === "function") onCollapse();
      wrapEl.classList.add("collapsed");
    } else {
      wrapEl.classList.remove("collapsed");
    }

    transitionTimer = setTimeout(finishTransition, 960);
  }

  if (wrapEl) {
    wrapEl.addEventListener("transitionend", (e) => {
      if (e.target === wrapEl && e.propertyName === "grid-template-rows") {
        finishTransition();
      }
    });
  }

  scrollEls.forEach((scrollEl) => {
    let pullStartY = 0;
    let scrollRafId = null;

    scrollEl.addEventListener(
      "touchstart",
      (e) => {
        if (scrollEl.offsetParent === null) return;
        activeScrollEl = scrollEl;
        pullStartY = e.touches[0].clientY;
      },
      { passive: true },
    );

    scrollEl.addEventListener(
      "touchmove",
      (e) => {
        if (scrollEl.offsetParent === null) return;
        if (
          isCollapsed &&
          scrollEl.scrollTop <= 5 &&
          e.touches[0].clientY - pullStartY > 25
        ) {
          activeScrollEl = scrollEl;
          setCollapsed(false);
        }
      },
      { passive: true },
    );

    scrollEl.addEventListener(
      "wheel",
      (e) => {
        if (scrollEl.offsetParent === null) return;
        if (isCollapsed && scrollEl.scrollTop <= 5 && e.deltaY < 0) {
          activeScrollEl = scrollEl;
          setCollapsed(false);
        }
      },
      { passive: true },
    );

    scrollEl.addEventListener(
      "scroll",
      () => {
        if (scrollEl.offsetParent === null || scrollRafId) return;
        scrollRafId = requestAnimationFrame(() => {
          scrollRafId = null;
          if (scrollEl.offsetParent === null) return;

          if (activeScrollEl !== scrollEl) {
            activeScrollEl = scrollEl;
            scrollAccum = 0;
          }

          const currentScroll = Math.max(0, scrollEl.scrollTop);
          const prevScroll = lastScrollMap.get(scrollEl) || 0;

          if (isTransitioning) {
            lastScrollMap.set(scrollEl, currentScroll);
            return;
          }

          if (currentScroll <= 6) {
            if (isCollapsed) setCollapsed(false);
            lastScrollMap.set(scrollEl, currentScroll);
            scrollAccum = 0;
            return;
          }

          const delta = currentScroll - prevScroll;
          lastScrollMap.set(scrollEl, currentScroll);
          if (Math.abs(delta) < 1) return;

          if (
            (delta > 0 && scrollAccum < 0) ||
            (delta < 0 && scrollAccum > 0)
          ) {
            scrollAccum = 0;
          }
          scrollAccum += delta;

          const scrollableRange = scrollEl.scrollHeight - scrollEl.clientHeight;
          const minNeededScroll = (wrapEl?.offsetHeight || 160) + 24;

          if (
            !isCollapsed &&
            scrollAccum > collapseAccum &&
            currentScroll > minScrollTop &&
            scrollableRange > minNeededScroll
          ) {
            setCollapsed(true);
          } else if (isCollapsed && scrollAccum < -expandAccum) {
            setCollapsed(false);
          }
        });
      },
      { passive: true },
    );
  });

  return { setCollapsed };
}

// --- SWIPE NAVIGATION LOGIC ---
(function initSwipeNavigation() {
  let touchStartX = 0;
  let touchEndX = 0;
  let touchStartY = 0;
  let touchEndY = 0;
  let touchStartTime = 0;
  let ignoreSwipe = false;

  const swipeArea = document.getElementById("dashboard-view");
  if (!swipeArea) return;

  function isInsideHorizontalScroller(target) {
    if (!(target instanceof Element)) return false;

    if (
      target.closest(
        ".dir-filter-bar, .dir-filter-chips, .quick-delete-selected-chips, .learn-flashcard-stage, #top-profile-btn, input, textarea, select",
      )
    ) {
      return true;
    }

    let el = target;
    while (el && el !== swipeArea) {
      const style = getComputedStyle(el);
      const canScrollX =
        (style.overflowX === "auto" || style.overflowX === "scroll") &&
        el.scrollWidth > el.clientWidth + 4;
      if (canScrollX) return true;
      el = el.parentElement;
    }
    return false;
  }

  swipeArea.addEventListener(
    "touchstart",
    (e) => {
      if (e.touches.length !== 1) {
        ignoreSwipe = true;
        return;
      }

      ignoreSwipe = isInsideHorizontalScroller(e.target);
      if (ignoreSwipe) return;

      touchStartX = e.changedTouches[0].screenX;
      touchStartY = e.changedTouches[0].screenY;
      touchStartTime = performance.now();
    },
    { passive: true },
  );

  swipeArea.addEventListener(
    "touchcancel",
    () => {
      ignoreSwipe = true;
    },
    { passive: true },
  );

  swipeArea.addEventListener(
    "touchend",
    (e) => {
      if (ignoreSwipe || e.changedTouches.length === 0) {
        ignoreSwipe = false;
        return;
      }

      touchEndX = e.changedTouches[0].screenX;
      touchEndY = e.changedTouches[0].screenY;
      const elapsed = Math.max(1, performance.now() - touchStartTime);

      if (
        document.getElementById("sidebar")?.classList.contains("open") ||
        document.querySelector(".modal-overlay.show") ||
        document
          .getElementById("directory-floating-bar")
          ?.classList.contains("open")
      ) {
        return;
      }

      const deltaX = touchEndX - touchStartX;
      const deltaY = touchEndY - touchStartY;
      const absX = Math.abs(deltaX);
      const absY = Math.abs(deltaY);
      const velocity = absX / elapsed; // px per ms

      // Hard-swipe thresholds: long distance + brisk flick + straight horizontal line
      const MIN_SWIPE_DISTANCE = Math.max(
        135,
        Math.round(window.innerWidth * 0.36),
      );
      const MAX_SWIPE_DURATION = 420;
      const MIN_SWIPE_VELOCITY = 0.45;
      const MAX_VERTICAL_DRIFT = 50;
      const HORIZONTAL_RATIO = 2.6;

      if (
        absX > MIN_SWIPE_DISTANCE &&
        elapsed <= MAX_SWIPE_DURATION &&
        velocity >= MIN_SWIPE_VELOCITY &&
        absY < MAX_VERTICAL_DRIFT &&
        absX > absY * HORIZONTAL_RATIO
      ) {
        const currentActiveId = viewOrder.find((view) =>
          document.getElementById("view-" + view)?.classList.contains("active"),
        );

        let currentIndex = viewOrder.indexOf(currentActiveId);
        if (currentIndex === -1) return;

        if (deltaX < 0 && currentIndex < viewOrder.length - 1) {
          switchDashboardView(viewOrder[currentIndex + 1]);
        } else if (deltaX > 0 && currentIndex > 0) {
          switchDashboardView(viewOrder[currentIndex - 1]);
        }
      }
    },
    { passive: true },
  );
})();

          once: true,

(function initSmartEnterNavigation() {
  const TEXT_INPUT_SELECTOR =
    'input:not([type="hidden"]):not([type="checkbox"]):not([type="radio"]):not([type="file"]):not([type="range"]):not([type="button"]):not([type="submit"])';

  const GROUP_CONTAINERS = [
    ".form-body",
    ".profile-edit-card",
    ".advanced-filters-inputs",
    "#approvals-filters-panel",
    "#manual-entries-container",
    ".edit-modal-card",
    "#office-modal-body",
  ].join(", ");

  document.addEventListener("keydown", function (e) {
    if (e.key !== "Enter" && e.keyCode !== 13) return;

    const currentInput = e.target;
    if (!currentInput || !currentInput.matches(TEXT_INPUT_SELECTOR)) return;

    e.preventDefault();

    if (currentInput.id === "search-input") {
      currentInput.blur();
      if (typeof triggerSearch === "function") triggerSearch(true);
      return;
    }

    if (currentInput.id === "fb-title") {
      const descEl = document.getElementById("fb-description");
      if (descEl) {
        descEl.focus();
        return;
      }
    }

    if (currentInput.id === "learn-pin-input") {
      if (typeof submitPinSniperAnswer === "function") submitPinSniperAnswer();
      return;
    }

    if (currentInput.id === "directory-filter-input") {
      currentInput.blur();
      if (typeof filterDirectory === "function")
        filterDirectory(true, { saveHistory: true });
      return;
    }

    if (
      currentInput.id === "members-filter-input" ||
      currentInput.id === "approvals-search-input" ||
      currentInput.id === "feedback-search-input"
    ) {
      currentInput.blur();
      if (
        currentInput.id === "approvals-search-input" &&
        typeof applyApprovalsFilters === "function"
      ) {
        applyApprovalsFilters(true);
      } else if (
        currentInput.id === "feedback-search-input" &&
        typeof applyFeedbackFilters === "function"
      ) {
        applyFeedbackFilters();
      }
      return;
    }

    if (typeof closeAllDropdowns === "function") {
      closeAllDropdowns();
    }

    const container = currentInput.closest(GROUP_CONTAINERS);
    if (container) {
      const visibleInputs = Array.from(
        container.querySelectorAll(TEXT_INPUT_SELECTOR),
      ).filter(
        (el) => el.offsetParent !== null && !el.disabled && !el.readOnly,
      );

      const currentIndex = visibleInputs.indexOf(currentInput);
      const nextInput = visibleInputs[currentIndex + 1];

      if (nextInput) {
        nextInput.focus();
        nextInput.scrollIntoView({ behavior: "smooth", block: "center" });
        return;
      }
    }

    currentInput.blur();
  });

  document.addEventListener("focusin", function (e) {
    const input = e.target;
    if (!input || input.tagName !== "INPUT") return;

    if (
      input.id === "search-input" ||
      input.id === "directory-filter-input" ||
      input.id === "members-filter-input" ||
      input.id === "feedback-search-input" ||
      input.id === "approvals-search-input"
    ) {
      input.setAttribute("enterkeyhint", "search");
      return;
    }

    if (input.id === "fb-title") {
      input.setAttribute("enterkeyhint", "next");
      return;
    }

    const container = input.closest(GROUP_CONTAINERS);
    if (!container) {
      input.setAttribute("enterkeyhint", "done");
      return;
    }

    const visibleInputs = Array.from(
      container.querySelectorAll(TEXT_INPUT_SELECTOR),
    ).filter((el) => el.offsetParent !== null && !el.disabled && !el.readOnly);

    const isLast = visibleInputs.indexOf(input) === visibleInputs.length - 1;
    input.setAttribute("enterkeyhint", isLast ? "done" : "next");
  });
})();

// --- DATA FETCHING & STATS ---
async function fetchAllPostOffices(forceSync = false) {
  if (!forceSync) {
    const cachedDataString = localStorage.getItem("dakmitra_offline_db");
    if (cachedDataString) {
      try {
        cachedPostOffices = JSON.parse(cachedDataString);
        initializeSearchAndRender();

        const runInBg =
          window.requestIdleCallback || ((cb) => setTimeout(cb, 300));
        runInBg(() => {
          fetchFromServerBackground();
          if (navigator.onLine) initRealtimeSync();
        });
        return true;
      } catch (e) {
        console.error("Cache corrupted, fetching fresh.");
      }
    }
  }

  return await fetchFromServerBackground();
}

async function fetchFromServerBackground() {
  if (!navigator.onLine) {
    console.log("Offline mode: Skipping background sync.");
    if (!cachedPostOffices || cachedPostOffices.length === 0) {
      document.getElementById("results-display-area").innerHTML =
        `<ul class="list-container"><li class="list-empty" style="color: #ef4444;">Offline mode active. No cached data available.</li></ul>`;
    }
    return false;
  }

  try {
    let allRows = [];
    let from = 0;
    const batchSize = 1000;

    while (true) {
      const { data, error } = await supabaseClient
        .from("post_offices")
        .select("*")
        .eq("is_deleted", false)
        .order("id", { ascending: true })
        .range(from, from + batchSize - 1);

      if (error) throw error;
      if (!data || data.length === 0) break;

      allRows = allRows.concat(data);
      if (data.length < batchSize) break;
      from += batchSize;
    }

    const newJson = JSON.stringify(allRows);
    const oldJson = localStorage.getItem("dakmitra_offline_db");

    if (newJson !== oldJson) {
      cachedPostOffices = allRows;
      localStorage.setItem("dakmitra_offline_db", newJson);
      initializeSearchAndRender(true);
    }
    return true;
  } catch (error) {
    console.error("Background sync failed:", error);
    return false;
  }
}

// --- SMART 3-DIGIT PIN SUFFIX OFFICE INDEX ---
let pinSuffixToOfficeMap = new Map();
window.activeCleanSearchQuery = "";
window.activeSmartOfficeFilter = null;

function buildPinSuffixIndex() {
  pinSuffixToOfficeMap.clear();
  if (!cachedPostOffices) return;

  cachedPostOffices.forEach((row) => {
    const pinStr = String(row.Pin || "").trim();
    const officeName = String(row.Office || "").trim();
    if (/^\d{6}$/.test(pinStr) && officeName) {
      const suffix = pinStr.slice(-3);
      if (!pinSuffixToOfficeMap.has(suffix)) {
        pinSuffixToOfficeMap.set(suffix, officeName);
      }
    }
  });
}

// --- PRECOMPUTED SEARCH & PRIORITY INDEX (ZERO-LAG MOBILE RANKING) ---
let searchIndexMap = new Map();

function createSearchMeta(row) {
  const addrRaw = String(row.Address || "").trim();
  const offRaw = String(row.Office || "").trim();
  const officeType = detectOfficeType(addrRaw);
  const isPriority = Number(row.priority) >= 1;
  const effectivePriority = getEffectivePriority(row);

  const addrCore = stripPostalSuffixes(addrRaw);
  const addrWords = addrCore ? addrCore.split(" ") : [];
  const addrWordPhonetics = [];
  for (let i = 0; i < addrWords.length; i++) {
    const pk = postalPhoneticKey(addrWords[i]);
    if (pk) addrWordPhonetics.push(pk);
  }

  const officeCore = stripPostalSuffixes(offRaw);
  const officeWords = officeCore ? officeCore.split(" ") : [];
  const officeWordPhonetics = [];
  for (let i = 0; i < officeWords.length; i++) {
    const pk = postalPhoneticKey(officeWords[i]);
    if (pk) officeWordPhonetics.push(pk);
  }

  return {
    officeType,
    isPriority,
    effectivePriority,
    addrLower: addrRaw.toLowerCase(),
    addrCore,
    addrWords,
    addrPhonetic: postalPhoneticKey(addrRaw),
    addrWordPhonetics,
    officeLower: offRaw.toLowerCase(),
    officeCore,
    officeWords,
    officePhonetic: postalPhoneticKey(offRaw),
    officeWordPhonetics,
  };
}

function buildSearchIndex() {
  searchIndexMap.clear();
  if (!cachedPostOffices) return;
  for (let i = 0; i < cachedPostOffices.length; i++) {
    const row = cachedPostOffices[i];
    if (row && row.id !== undefined && row.id !== null) {
      searchIndexMap.set(String(row.id), createSearchMeta(row));
    }
  }
}

function updateSearchIndexRow(row) {
  if (!row || row.id === undefined || row.id === null) return;
  searchIndexMap.set(String(row.id), createSearchMeta(row));
}

function getSearchMeta(row) {
  if (!row) return null;
  const key = String(row.id);
  let meta = searchIndexMap.get(key);
  if (!meta) {
    meta = createSearchMeta(row);
    if (row.id !== undefined && row.id !== null) {
      searchIndexMap.set(key, meta);
    }
  }
  return meta;
}

function compareByPriorityOnly(aRow, bRow) {
  const aMeta = getSearchMeta(aRow);
  const bMeta = getSearchMeta(bRow);
  const prioDiff =
    (bMeta?.effectivePriority || 0) - (aMeta?.effectivePriority || 0);
  if (prioDiff !== 0) return prioDiff;
  return (aMeta?.addrLower || "").localeCompare(bMeta?.addrLower || "");
}

function parseSmartSearchQuery(
  rawInput,
  searchMode,
  manualOfficeFilter,
  manualPinFilter,
) {
  let text = String(rawInput || "").trim();

  if (!text || manualOfficeFilter || manualPinFilter) {
    return { cleanQuery: text, detectedSuffix: null, targetOffice: null };
  }

  if (pinSuffixToOfficeMap.size === 0 && cachedPostOffices?.length > 0) {
    buildPinSuffixIndex();
  }

  text = text
    .replace(/(?:^|\s)(\d)\s+(\d)\s+(\d)$/, " $1$2$3")
    .replace(/^(\d)\s+(\d)\s+(\d)(?:\s|$)/, "$1$2$3 ")
    .trim();

  // Standalone 3-digit unique PIN suffix (e.g. "501" or voice "5 0 1" -> Kachhwa SO / 231501)
  if (/^\d{3}$/.test(text) && pinSuffixToOfficeMap.has(text)) {
    return {
      cleanQuery: "",
      detectedSuffix: text,
      targetOffice: pinSuffixToOfficeMap.get(text),
    };
  }

  if (searchMode === "pin") {
    return { cleanQuery: text, detectedSuffix: null, targetOffice: null };
  }

  const patterns = [
    { regex: /^(.*?)\s*[@\-/]\s*(\d{3})\s*$/, queryIdx: 1, suffixIdx: 2 },
    { regex: /^(.+?[a-zA-Z])\s*(\d{3})$/, queryIdx: 1, suffixIdx: 2 },
    { regex: /^(\d{3})\s*([a-zA-Z].*)$/, queryIdx: 2, suffixIdx: 1 },
  ];

  for (const { regex, queryIdx, suffixIdx } of patterns) {
    const match = text.match(regex);
    if (match && pinSuffixToOfficeMap.has(match[suffixIdx])) {
      const suffix = match[suffixIdx];
      return {
        cleanQuery: match[queryIdx].trim(),
        detectedSuffix: suffix,
        targetOffice: pinSuffixToOfficeMap.get(suffix),
      };
    }
  }

  return { cleanQuery: text, detectedSuffix: null, targetOffice: null };
}

let directoryGroupByKeyMap = new Map();

function buildDirectoryGroupedData() {
  directoryGroupByKeyMap.clear();
  if (!cachedPostOffices || cachedPostOffices.length === 0) {
    directoryGroupedData = [];
    return;
  }
  const grouped = {};
  cachedPostOffices.forEach((row) => {
    const rawOffice = String(row.Office || "Unknown Office")
      .trim()
      .replace(/\s+/g, " ");
    const officeKey = rawOffice.toLowerCase();

    if (!grouped[officeKey]) {
      grouped[officeKey] = {
        office: rawOffice,
        pin: row.Pin || "N/A",
        addresses: [],
      };
    }

    grouped[officeKey].addresses.push({
      id: row.id,
      Address: row.Address || "",
      Office: grouped[officeKey].office,
      Pin: row.Pin || "",
      officeType: detectOfficeType(row.Address),
      priority: Number(row.priority) >= 1 ? 1 : 0,
      effectivePriority: getEffectivePriority(row),
    });
  });

  directoryGroupedData = Object.values(grouped)
    .sort((a, b) => a.office.localeCompare(b.office))
    .map((group) => {
      const built = {
        office: group.office,
        pin: group.pin,
        addresses: group.addresses.sort(
          (a, b) =>
            b.effectivePriority - a.effectivePriority ||
            String(a.Address).localeCompare(String(b.Address)),
        ),
        searchString: (group.office + " " + group.pin).toLowerCase(),
      };
      directoryGroupByKeyMap.set(group.office.toLowerCase(), built);
      return built;
    });
}

function syncFavoritesWithCache() {
  const favorites = getSafeStorage(FAV_KEY);
  if (!favorites.length) return;

  const byId = new Map(cachedPostOffices.map((r) => [String(r.id), r]));
  let changed = false;

  const updatedFavs = [];
  favorites.forEach((fav) => {
    const live = byId.get(String(fav.id));
    if (!live) {
      changed = true;
      return;
    }
    const nextPin = live.Pin ?? "";
    if (
      fav.address !== live.Address ||
      fav.office !== live.Office ||
      String(fav.pin ?? "") !== String(nextPin)
    ) {
      changed = true;
      updatedFavs.push({
        id: live.id,
        address: live.Address,
        office: live.Office,
        pin: nextPin,
      });
    } else {
      updatedFavs.push(fav);
    }
  });

  if (changed) {
    localStorage.setItem(FAV_KEY, JSON.stringify(updatedFavs));
    if (
      document.getElementById("view-favorites")?.classList.contains("active")
    ) {
      renderFavorites();
    }
  }
}

function isRowDataUnchanged(existing, incoming) {
  if (!existing || !incoming) return false;
  const keys = Object.keys(incoming);
  for (let i = 0; i < keys.length; i++) {
    const k = keys[i];
    const oldVal = existing[k];
    const newVal = incoming[k];
    if (oldVal === newVal) continue;
    if (
      oldVal !== null &&
      newVal !== null &&
      typeof oldVal === "object" &&
      typeof newVal === "object"
    ) {
      if (JSON.stringify(oldVal) === JSON.stringify(newVal)) continue;
    }
    return false;
  }
  return true;
}

function upsertLocalPostOffice(row) {
  if (!row || row.id === undefined || row.id === null) return;

  if (row.is_deleted === true) {
    removeLocalPostOffice(row.id);
    return;
  }

  const idx = cachedPostOffices.findIndex(
    (r) => String(r.id) === String(row.id),
  );
  if (idx > -1) {
    if (isRowDataUnchanged(cachedPostOffices[idx], row)) {
      return;
    }
    cachedPostOffices[idx] = { ...cachedPostOffices[idx], ...row };
    updateSearchIndexRow(cachedPostOffices[idx]);
  } else {
    cachedPostOffices.push(row);
    updateSearchIndexRow(row);
  }

  if (
    typeof activeOfficeRecord !== "undefined" &&
    activeOfficeRecord &&
    String(activeOfficeRecord.id) === String(row.id)
  ) {
    activeOfficeRecord =
      cachedPostOffices[idx > -1 ? idx : cachedPostOffices.length - 1];
    const isModalOpen = document
      .getElementById("office-details-modal-overlay")
      ?.classList.contains("show");
    const isViewMode = !document
      .getElementById("office-modal-view-actions")
      ?.classList.contains("hidden");
    if (
      isModalOpen &&
      isViewMode &&
      typeof renderOfficeStaffViewMode === "function"
    ) {
      renderOfficeStaffViewMode();
    }
  }

  persistPostOfficeCache();
  scheduleLocalUiRefresh();
}

function removeLocalPostOffice(id) {
  if (id === undefined || id === null) return;

  const beforeLen = cachedPostOffices.length;
  cachedPostOffices = cachedPostOffices.filter(
    (r) => String(r.id) !== String(id),
  );
  if (cachedPostOffices.length === beforeLen) return;

  if (
    typeof activeOfficeRecord !== "undefined" &&
    activeOfficeRecord &&
    String(activeOfficeRecord.id) === String(id)
  ) {
    if (typeof closeOfficeDetailsModal === "function") {
      closeOfficeDetailsModal();
    }
  }

  persistPostOfficeCache();
  scheduleLocalUiRefresh();
}

}

// --- SUPABASE REALTIME SYNC ---
let realtimeChannel = null;

function initRealtimeSync() {
  if (!supabaseClient || realtimeChannel) return;

  realtimeChannel = supabaseClient
    .channel("dakmitra-live-sync")
    .on(
      "postgres_changes",
      { event: "INSERT", schema: "public", table: "post_offices" },
      (payload) => {
        upsertLocalPostOffice(payload.new);
      },
    )
    .on(
      "postgres_changes",
      { event: "UPDATE", schema: "public", table: "post_offices" },
      (payload) => {
        upsertLocalPostOffice(payload.new);
      },
    )
    .on(
      "postgres_changes",
      { event: "DELETE", schema: "public", table: "post_offices" },
      (payload) => {
        if (payload.old?.id) removeLocalPostOffice(payload.old.id);
      },
    )
    .on(
      "postgres_changes",
      { event: "*", schema: "public", table: "edit_requests" },
      () => {
        if (typeof scheduleApprovalsRealtimeRefresh === "function") {
          scheduleApprovalsRealtimeRefresh();
        } else {
          invalidateApprovalsCache();
          if (
            document
              .getElementById("view-approvals")
              ?.classList.contains("active")
          ) {
            switchApprovalTab(currentApprovalTab, true);
          }
        }
      },
    )
    .on(
      "postgres_changes",
      { event: "*", schema: "public", table: "request_approvals" },
      () => {
        if (typeof scheduleApprovalsRealtimeRefresh === "function") {
          scheduleApprovalsRealtimeRefresh();
        }
      },
    )
    .on(
      "postgres_changes",
      { event: "*", schema: "public", table: "feedback_items" },
      () => {

          scheduleFeedbackRealtimeRefresh();

    )

// --- FULL OFFLINE-TO-ONLINE HYDRATION ENGINE ---
let wasOffline = !navigator.onLine;
let isHydratingOnline = false;
let onlineRetryTimer = null;

window.addEventListener("offline", () => {
  wasOffline = true;
  clearTimeout(onlineRetryTimer);
  teardownRealtimeSync();
  showOfflineToast("Switched to offline cache.");
});

async function hydrateOnlineState(attempt = 1) {
  if (isHydratingOnline) return;
  clearTimeout(onlineRetryTimer);

  const hasInternet = await checkActualInternet();
  if (!hasInternet) {
    if (attempt <= 4 && navigator.onLine) {
      const delays = [1500, 3000, 5000, 8000];
      onlineRetryTimer = setTimeout(
        () => hydrateOnlineState(attempt + 1),
        delays[attempt - 1],
      );
    }
    return;
  }

  isHydratingOnline = true;

  try {
    const client = await ensureSupabaseReady();
    if (!client) throw new Error("Supabase client failed to initialize.");

    let {
      data: { session },
    } = await client.auth.getSession();

    const nowSec = Math.floor(Date.now() / 1000);
    if (session && (!session.expires_at || session.expires_at - nowSec < 120)) {
      const { data: refreshed, error: refreshErr } =
        await client.auth.refreshSession();
      if (!refreshErr && refreshed?.session) {
        session = refreshed.session;
      }
    }

    if (!session || !session.user) {
      wasOffline = false;
      if (typeof checkUserSession === "function") checkUserSession();
      return;
    }

    currentUserId = session.user.id;

    isCurrentUserAdmin = await checkIfAdmin(session.user.id);
    applyAdminUI();

    const { data: profileData } = await client
      .from("profiles")
      .select("name, designation, employee_id, mobile, avatar_url, show_mobile")
      .eq("id", session.user.id)
      .maybeSingle();

    if (profileData) {
      const merged = {
        ...profileData,
        email: session.user.email,
        is_admin: isCurrentUserAdmin,
      };
      localStorage.setItem("dakmitra_cached_profile", JSON.stringify(merged));
      renderUserProfileUI(merged, session.user.email);
    }

    teardownRealtimeSync();
    if (client.realtime && session.access_token) {
      client.realtime.setAuth(session.access_token);
    }
    initRealtimeSync();

    invalidateApprovalsCache();

    if (typeof invalidateFeedbackCache === "function") {
      invalidateFeedbackCache();
    }
    if (typeof refreshFeedbackOpenCount === "function") {
      refreshFeedbackOpenCount();
    }
    if (typeof syncLearningProgressInBackground === "function") {
      syncLearningProgressInBackground();
    }
    if (
      document.getElementById("view-feedback")?.classList.contains("active") &&
      typeof loadFeedbackBoardOrCache === "function"
    ) {
      loadFeedbackBoardOrCache(true);
    }

    if (
      document.getElementById("view-approvals")?.classList.contains("active")
    ) {
      switchApprovalTab(currentApprovalTab, true);
    }

    await fetchFromServerBackground();

    if (wasOffline) {
      wasOffline = false;
      showToast("ðŸŒ Back online â€” synced!");
    }
  } catch (err) {
    console.warn("Online hydration failed:", err);
    if (attempt <= 3 && navigator.onLine) {
      onlineRetryTimer = setTimeout(
        () => hydrateOnlineState(attempt + 1),
        3000,
      );
    }
  } finally {
    isHydratingOnline = false;
  }
}

window.addEventListener("online", () => {
  hydrateOnlineState(1);
});

document.addEventListener("visibilitychange", () => {
  if (document.visibilityState === "visible" && navigator.onLine) {
    if (wasOffline || !supabaseClient || !realtimeChannel) {
      hydrateOnlineState(1);
    } else {
      const isLoggedIn = !document
        .getElementById("dashboard-view")
        ?.classList.contains("hidden");
      if (isLoggedIn) fetchFromServerBackground();
    }
  }
});

async function handleAuth() {
  await ensureSupabaseReady();
  hideAlert();
  const email = document.getElementById("email").value.trim();
  const password = document.getElementById("password").value.trim();
  if (!email || !password)
    return showAlert("Please fill in all required fields.");

  const splashScreen = document.getElementById("splash-screen");
  splashScreen.classList.remove("hidden-splash");

  if (currentMode === "signup") {
    const name = document.getElementById("signup-name").value.trim();
    const employeeId = document.getElementById("signup-empid").value.trim();
    const mobile = document.getElementById("signup-mobile").value.trim();
    const designation = document
      .getElementById("signup-designation")
      .value.trim();

    if (!name || !employeeId || !mobile || !designation) {
      splashScreen.classList.add("hidden-splash");
      return showAlert("Please complete all profile fields.");
    }

    if (!/^\d{10}$/.test(mobile)) {
      splashScreen.classList.add("hidden-splash");
      return showAlert("Please enter a valid 10-digit mobile number.");
    }

    const { error: authError } = await supabaseClient.auth.signUp({
      email,
      password,
      options: {
        data: { name, employee_id: employeeId, mobile, designation },
        emailRedirectTo: window.location.origin + window.location.pathname,
      },
    });

    splashScreen.classList.add("hidden-splash");
    if (authError) return showAlert(authError.message);

    showToast("âœ‰ï¸ Account created!");
    showAlert(
      "Registration successful! Check your email to verify.",
      "success",
    );
    switchTab("signin");
  } else {
    const { data: authData, error: authError } =
      await supabaseClient.auth.signInWithPassword({
        email,
        password,
      });

    if (authError) {
      splashScreen.classList.add("hidden-splash");
      return showAlert(authError.message);
    }

    let localDeviceId = localStorage.getItem("dakmitra_device_id");
    if (!localDeviceId) {
      localDeviceId =
        "dev_" + Math.random().toString(36).substring(2) + Date.now();
      localStorage.setItem("dakmitra_device_id", localDeviceId);
    }

    const { data: profileData, error: profileError } = await supabaseClient
      .from("profiles")
      .select(
        "device_id, name, designation, employee_id, mobile, avatar_url, show_mobile",
      )
      .eq("id", authData.user.id)
      .single();

    if (profileError) {
      await supabaseClient.auth.signOut();
      splashScreen.classList.add("hidden-splash");
      return showAlert("Error checking device registration.");
    }

    if (profileData.device_id && profileData.device_id !== localDeviceId) {
      await supabaseClient.auth.signOut();
      splashScreen.classList.add("hidden-splash");
      return showAlert(
        "Login Denied: Account already registered on another device.",
      );
    }

    if (!profileData.device_id) {
      await supabaseClient
        .from("profiles")
        .update({ device_id: localDeviceId })
        .eq("id", authData.user.id);
    }

    localStorage.setItem(
      "dakmitra_cached_profile",
      JSON.stringify({ ...profileData, email: authData.user.email }),
    );
    checkUserSession();
  }
}

const cropState = {
  img: null,
  natW: 0,
  natH: 0,
  scale: 1,
  minScale: 1,
  maxScale: 5,
  tx: 0,
  ty: 0,
  rotation: 0,
  guideShape: "round",
  cropSize: 300,
  previewSquareSize: 50,
  previewRoundSize: 48,
  pointers: new Map(),
  lastPinchDist: 0,
  lastPinchMid: null,
  lastTapTime: 0,
  objectUrl: null,
};

function getCropSize() {
  const cutout = document.getElementById("crop-cutout");
  return cutout ? cutout.getBoundingClientRect().width || 300 : 300;
}

function syncCropMetrics() {
  cropState.cropSize = getCropSize();
  const sqFrame = document.getElementById("crop-preview-square-frame");
  const rdFrame = document.getElementById("crop-preview-round-frame");
  if (sqFrame && sqFrame.clientWidth > 0) {
    cropState.previewSquareSize = sqFrame.clientWidth;
  }
  if (rdFrame && rdFrame.clientWidth > 0) {
    cropState.previewRoundSize = rdFrame.clientWidth;
  }
}

function setCropGuideShape(shape) {
  cropState.guideShape = shape === "square" ? "square" : "round";
  const cutout = document.getElementById("crop-cutout");
  const sqBtn = document.getElementById("crop-preview-square-btn");
  const rdBtn = document.getElementById("crop-preview-round-btn");

  cutout?.classList.toggle("is-square", cropState.guideShape === "square");
  sqBtn?.classList.toggle("active", cropState.guideShape === "square");
  rdBtn?.classList.toggle("active", cropState.guideShape === "round");

  if (navigator.vibrate) navigator.vibrate(20);
}

function computeCropBounds() {
  const cropSize = cropState.cropSize || getCropSize();
  const swapped = cropState.rotation % 180 !== 0;
  const baseW = swapped ? cropState.natH : cropState.natW;
  const baseH = swapped ? cropState.natW : cropState.natH;

  cropState.minScale = Math.max(cropSize / baseW, cropSize / baseH);
  cropState.maxScale = cropState.minScale * 4.5;
  cropState.scale = Math.min(
    Math.max(cropState.scale, cropState.minScale),
    cropState.maxScale,
  );

  const effW = baseW * cropState.scale;
  const effH = baseH * cropState.scale;
  const maxTx = Math.max(0, (effW - cropSize) / 2);
  const maxTy = Math.max(0, (effH - cropSize) / 2);

  cropState.tx = Math.min(Math.max(cropState.tx, -maxTx), maxTx);
  cropState.ty = Math.min(Math.max(cropState.ty, -maxTy), maxTy);
}

function applyCropTransform(animate = false) {
  const imgEl = document.getElementById("crop-source-img");
  if (!imgEl) return;

  imgEl.classList.toggle("is-animating", animate);
  imgEl.style.transform = `translate3d(calc(-50% + ${cropState.tx}px), calc(-50% + ${cropState.ty}px), 0) scale(${cropState.scale}) rotate(${cropState.rotation}deg)`;

  // Sync both live previews at 60fps using exact scaled GPU transforms
  const cropSize = cropState.cropSize || getCropSize() || 300;
  const previews = [
    {
      el: document.getElementById("crop-preview-square-img"),
      size: cropState.previewSquareSize,
    },
    {
      el: document.getElementById("crop-preview-round-img"),
      size: cropState.previewRoundSize,
    },
  ];

  previews.forEach(({ el, size }) => {
    if (!el || !size) return;
    const ratio = size / cropSize;
    el.classList.toggle("is-animating", animate);
    el.style.transform = `translate3d(calc(-50% + ${
      cropState.tx * ratio
    }px), calc(-50% + ${cropState.ty * ratio}px), 0) scale(${
      cropState.scale * ratio
    }) rotate(${cropState.rotation}deg)`;
  });
}

function handleProfilePicUpload(event) {
  const file = event.target.files?.[0];
  if (!file) return;
  if (!file.type.startsWith("image/")) {
    return showToast("âš ï¸ Please select a valid image file.");
  }

  if (cropState.objectUrl) URL.revokeObjectURL(cropState.objectUrl);
  cropState.objectUrl = URL.createObjectURL(file);

  const imgEl = document.getElementById("crop-source-img");
  const sqPrevImg = document.getElementById("crop-preview-square-img");
  const rdPrevImg = document.getElementById("crop-preview-round-img");

  const loader = new Image();
  loader.onload = () => {
    cropState.img = loader;
    cropState.natW = loader.naturalWidth;
    cropState.natH = loader.naturalHeight;
    cropState.rotation = 0;
    cropState.tx = 0;
    cropState.ty = 0;

    [imgEl, sqPrevImg, rdPrevImg].forEach((el) => {
      if (!el) return;
      el.src = cropState.objectUrl;
      el.style.width = `${cropState.natW}px`;
      el.style.height = `${cropState.natH}px`;
    });

    setCropGuideShape("round");
    document.getElementById("crop-modal-overlay").classList.add("show");

    requestAnimationFrame(() => {
      syncCropMetrics();
      cropState.scale = 0;
      computeCropBounds();
      applyCropTransform(false);
    });
  };
  loader.src = cropState.objectUrl;
}

function rotateCropImage() {
  cropState.rotation = (cropState.rotation + 90) % 360;
  computeCropBounds();
  applyCropTransform(true);
}

function renderCropToCanvas() {
  const canvas = document.getElementById("crop-canvas");
  if (!canvas || !cropState.img) return null;

  const outSize = canvas.width;
  const ratio = outSize / getCropSize();
  const ctx = canvas.getContext("2d");

  ctx.clearRect(0, 0, outSize, outSize);
  ctx.imageSmoothingEnabled = true;
  ctx.imageSmoothingQuality = "high";
  ctx.save();
  ctx.translate(outSize / 2, outSize / 2);
  ctx.translate(cropState.tx * ratio, cropState.ty * ratio);
  ctx.scale(cropState.scale * ratio, cropState.scale * ratio);
  ctx.rotate((cropState.rotation * Math.PI) / 180);
  ctx.drawImage(
    cropState.img,
    -cropState.natW / 2,
    -cropState.natH / 2,
    cropState.natW,
    cropState.natH,
  );
  ctx.restore();
  return canvas;
}

(function initCropGestures() {
  const stage = document.getElementById("crop-stage");
  if (!stage) return;

  function getPinchMetrics() {
    const pts = Array.from(cropState.pointers.values());
    const dx = pts[0].x - pts[1].x;
    const dy = pts[0].y - pts[1].y;
    return {
      dist: Math.hypot(dx, dy),
      mid: { x: (pts[0].x + pts[1].x) / 2, y: (pts[0].y + pts[1].y) / 2 },
    };
  }

  stage.addEventListener("pointerdown", (e) => {
    stage.setPointerCapture(e.pointerId);
    cropState.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
    stage.classList.add("is-interacting");

    if (cropState.pointers.size === 1) {
      const now = Date.now();
      if (now - cropState.lastTapTime < 280) {
        const targetScale =
          cropState.scale > cropState.minScale * 1.15
            ? cropState.minScale
            : cropState.minScale * 2.2;
        cropState.scale = targetScale;
        if (targetScale === cropState.minScale) {
          cropState.tx = 0;
          cropState.ty = 0;
        }
        computeCropBounds();
        applyCropTransform(true);
        cropState.lastTapTime = 0;
      } else {
        cropState.lastTapTime = now;
      }
    } else if (cropState.pointers.size === 2) {
      const { dist, mid } = getPinchMetrics();
      cropState.lastPinchDist = dist;
      cropState.lastPinchMid = mid;
    }
  });

  stage.addEventListener("pointermove", (e) => {
    if (!cropState.pointers.has(e.pointerId)) return;
    const prev = cropState.pointers.get(e.pointerId);
    const curr = { x: e.clientX, y: e.clientY };
    cropState.pointers.set(e.pointerId, curr);

    if (cropState.pointers.size === 1) {
      cropState.tx += curr.x - prev.x;
      cropState.ty += curr.y - prev.y;
      computeCropBounds();
      applyCropTransform(false);
    } else if (cropState.pointers.size === 2) {
      const { dist, mid } = getPinchMetrics();
      if (cropState.lastPinchDist > 0) {
        const rect = stage.getBoundingClientRect();
        const cx = rect.left + rect.width / 2;
        const cy = rect.top + rect.height / 2;

        const oldScale = cropState.scale;
        const newScale = Math.min(
          Math.max(
            oldScale * (dist / cropState.lastPinchDist),
            cropState.minScale,
          ),
          cropState.maxScale,
        );
        const factor = newScale / oldScale;

        const mx = mid.x - cx;
        const my = mid.y - cy;
        cropState.tx =
          mx -
          (mx - cropState.tx) * factor +
          (mid.x - cropState.lastPinchMid.x);
        cropState.ty =
          my -
          (my - cropState.ty) * factor +
          (mid.y - cropState.lastPinchMid.y);
        cropState.scale = newScale;

        computeCropBounds();
        applyCropTransform(false);
      }
      cropState.lastPinchDist = dist;
      cropState.lastPinchMid = mid;
    }
  });

  function handlePointerEnd(e) {
    cropState.pointers.delete(e.pointerId);
    if (cropState.pointers.size < 2) {
      cropState.lastPinchDist = 0;
      cropState.lastPinchMid = null;
    }
    if (cropState.pointers.size === 0) {
      stage.classList.remove("is-interacting");
    }
  }

  stage.addEventListener("pointerup", handlePointerEnd);
  stage.addEventListener("pointercancel", handlePointerEnd);

  stage.addEventListener(
    "wheel",
    (e) => {
      e.preventDefault();
      const zoomFactor = e.deltaY < 0 ? 1.1 : 0.91;
      cropState.scale *= zoomFactor;
      computeCropBounds();
      applyCropTransform(false);
    },
    { passive: false },
  );

  window.addEventListener("resize", () => {
    if (
      cropState.img &&
      document.getElementById("crop-modal-overlay")?.classList.contains("show")
    ) {
      syncCropMetrics();
      computeCropBounds();
      applyCropTransform(false);
    }
  });
})();

function applyLightboxTransform(dragging = false) {
  const overlay = document.getElementById("avatar-lightbox-overlay");
  const frame = document.getElementById("avatar-lightbox-frame");
  if (!frame) return;

  frame.classList.toggle("is-dragging", dragging);

  if (lbState.scale === 1 && lbState.ty > 0) {
    const progress = Math.min(lbState.ty / 260, 1);
    const shrink = 1 - progress * 0.18;
    frame.style.transform = `translate3d(${lbState.tx * 0.35}px, ${
      lbState.ty
    }px, 0) scale(${shrink})`;
    if (overlay) {

        0.96 - progress * 0.65

    frame.style.transform = `translate3d(${lbState.tx}px, ${lbState.ty}px, 0) scale(${lbState.scale})`;
    if (overlay) overlay.style.backgroundColor = "";
  }
}

(function initLightboxGestures() {
  const stage = document.getElementById("lightbox-stage");
  const shell = document.getElementById("lightbox-shell");
  if (!stage) return;

  function clampLightboxPan() {
    const frame = document.getElementById("avatar-lightbox-frame");
    if (!frame) return;
    const maxOffset = (frame.offsetWidth * (lbState.scale - 1)) / 2;
    lbState.tx = Math.min(Math.max(lbState.tx, -maxOffset), maxOffset);
    lbState.ty = Math.min(Math.max(lbState.ty, -maxOffset), maxOffset);
  }

  stage.addEventListener("pointerdown", (e) => {
    stage.setPointerCapture(e.pointerId);
    lbState.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
    lbState.moved = false;

    if (lbState.pointers.size === 2) {
      const pts = Array.from(lbState.pointers.values());
      lbState.lastPinchDist = Math.hypot(
        pts[0].x - pts[1].x,
        pts[0].y - pts[1].y,
      );
    }
  });

  stage.addEventListener("pointermove", (e) => {
    if (!lbState.pointers.has(e.pointerId)) return;
    const prev = lbState.pointers.get(e.pointerId);
    const curr = { x: e.clientX, y: e.clientY };
    lbState.pointers.set(e.pointerId, curr);

    const dx = curr.x - prev.x;
    const dy = curr.y - prev.y;
    if (Math.hypot(dx, dy) > 2) lbState.moved = true;

    if (lbState.pointers.size === 1) {
      if (lbState.scale > 1) {
        lbState.tx += dx;
        lbState.ty += dy;
        clampLightboxPan();
        applyLightboxTransform(true);
      } else {
        lbState.tx += dx;
        lbState.ty = Math.max(0, lbState.ty + dy);
        applyLightboxTransform(true);
      }
    } else if (lbState.pointers.size === 2 && lbState.hasPhoto) {
      const pts = Array.from(lbState.pointers.values());
      const dist = Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y);
      if (lbState.lastPinchDist > 0) {
        lbState.scale = Math.min(
          Math.max(lbState.scale * (dist / lbState.lastPinchDist), 1),
          4,
        );
        clampLightboxPan();
        applyLightboxTransform(true);
      }
      lbState.lastPinchDist = dist;
    }
  });

  function handleLbEnd(e) {
    lbState.pointers.delete(e.pointerId);
    if (lbState.pointers.size < 2) lbState.lastPinchDist = 0;
    if (lbState.pointers.size > 0) return;

    if (!lbState.moved) {
      const now = Date.now();
      if (now - lbState.lastTapTime < 260) {
        clearTimeout(lbState.tapTimer);
        lbState.lastTapTime = 0;
        if (lbState.hasPhoto) {
          lbState.scale = lbState.scale > 1.1 ? 1 : 2.4;
          lbState.tx = 0;
          lbState.ty = 0;
          applyLightboxTransform(false);
        }
      } else {
        lbState.lastTapTime = now;
        lbState.tapTimer = setTimeout(() => {
          shell?.classList.toggle("ui-hidden");
        }, 260);
      }
      return;
    }

    if (lbState.scale <= 1.02) {
      if (lbState.ty > 95) {
        closeProfilePhotoModal();
      } else {
        lbState.scale = 1;
        lbState.tx = 0;
        lbState.ty = 0;
        applyLightboxTransform(false);
      }
    }
  }

  stage.addEventListener("pointerup", handleLbEnd);
  stage.addEventListener("pointercancel", handleLbEnd);

  document.addEventListener("keydown", (e) => {
    if (e.key === "Escape") {
      if (
        document
          .getElementById("avatar-lightbox-overlay")
          ?.classList.contains("show")
      ) {
        closeProfilePhotoModal();
      } else if (
        document
          .getElementById("crop-modal-overlay")
          ?.classList.contains("show")
      ) {
        closeCropModal();
      }
    }
  });
})();

async function renderMembersDirectory(forceRefresh = false) {
  const contentEl = document.getElementById("members-content");
  if (!contentEl) return;

  // 1. Instant render from local cache if available
  if (membersDirectoryData.length === 0) {
    try {
      const cached = JSON.parse(localStorage.getItem(MEMBERS_CACHE_KEY));
      if (Array.isArray(cached) && cached.length > 0) {
        membersDirectoryData = cached;
      }
    } catch (_) {}
  }

  if (membersDirectoryData.length > 0) {
    renderMembersListUI();
  } else {
    contentEl.innerHTML = `
      <div class="skeleton-card"><div class="skeleton-line medium"></div><div class="skeleton-line short"></div></div>
      <div class="skeleton-card"><div class="skeleton-line medium"></div><div class="skeleton-line short"></div></div>
      <div class="skeleton-card"><div class="skeleton-line medium"></div><div class="skeleton-line short"></div></div>`;
  }

  if (membersDirectoryLoaded && !forceRefresh) return;

  // 2. Refresh fresh list from Supabase RPC
  const handleOfflineEmployees = () => {
    if (membersDirectoryData.length === 0) {
      contentEl.innerHTML = `<div class="list-empty">ðŸ“¡ No internet connection and no cached employees saved yet.</div>`;
    } else {
      showOfflineToast("Showing cached employees.");
    }
  };

  if (!navigator.onLine) {
    handleOfflineEmployees();
    return;
  }

  try {
    await ensureSupabaseReady();
    const { data, error } = await supabaseClient.rpc("get_members_directory");
    if (error) throw error;

    membersDirectoryData = Array.isArray(data) ? data : [];
    membersDirectoryLoaded = true;
    try {
      localStorage.setItem(
        MEMBERS_CACHE_KEY,
        JSON.stringify(membersDirectoryData),
      );
    } catch (_) {}

    renderMembersListUI();
  } catch (err) {
    if (isNetworkError(err)) {
      handleOfflineEmployees();
      return;
    }
    if (membersDirectoryData.length === 0) {
      contentEl.innerHTML = `<div class="list-empty" style="color: #ef4444;">Failed to load employees: ${escapeHTML(
        err.message,
      )}</div>`;
    }
  }
}

async function openUserProfileModal(userId, fallbackName = "") {
  if (!userId) return;

  const overlay = document.getElementById("user-profile-modal-overlay");
  const bodyEl = document.getElementById("user-modal-body");
  const actionsEl = document.getElementById("user-modal-actions");
  if (!overlay || !bodyEl) return;

  const cacheKey = String(userId);
  const cachedEntry = userProfileCache.get(cacheKey);
  if (cachedEntry && Date.now() - cachedEntry.ts < USER_PROFILE_CACHE_TTL_MS) {
    renderUserProfileModalContent(cachedEntry.data);
    overlay.classList.add("show");
    return;
  }

  // Show immediate loading skeleton inside modal
  const tempInitial = (fallbackName || "U").trim().charAt(0).toUpperCase();
  bodyEl.innerHTML = `
    <div class="user-modal-banner"></div>
    <div class="user-modal-avatar-wrap">
      <div class="user-modal-avatar">${escapeHTML(tempInitial)}</div>
    </div>
    <div style="padding: 0 20px 24px;">
      <h3 style="font-size: 19px; font-weight: 800; color: var(--text); margin-bottom: 12px;">
        ${escapeHTML(fallbackName || "Loading profile...")}
      </h3>
      <div class="skeleton-line medium" style="margin: 0 auto 10px;"></div>
      <div class="skeleton-line short" style="margin: 0 auto;"></div>
    </div>
  `;
  if (actionsEl) {
    actionsEl.innerHTML = `<button type="button" class="dm-btn dm-btn-secondary dm-btn-full" onclick="closeUserProfileModal()">Close</button>`;
  }
  overlay.classList.add("show");

  const handleOfflineProfile = () => {
    const basicMember = membersDirectoryData.find(
      (m) => String(m.id) === cacheKey,
    );
    if (basicMember) {
      renderUserProfileModalContent(basicMember, true);
      showOfflineToast("Showing cached basic profile.");
    } else {
      bodyEl.innerHTML = `<div style="padding: 40px 20px; color: var(--text-light); font-size: 14px;">ðŸ“¡ Connect to the internet to load this user's profile.</div>`;
    }
  };

  if (!navigator.onLine) {
    handleOfflineProfile();
    return;
  }

  try {
    await ensureSupabaseReady();
    const { data, error } = await supabaseClient.rpc(
      "get_user_public_profile",
      { target_user_id: userId },
    );
    if (error) throw error;

    const profile = Array.isArray(data) ? data[0] : data;
    if (!profile) {
      bodyEl.innerHTML = `<div style="padding: 40px 20px; color: var(--text-light); font-size: 14px;">Profile not found.</div>`;
      return;
    }

    userProfileCache.set(cacheKey, { data: profile, ts: Date.now() });
    renderUserProfileModalContent(profile);
  } catch (err) {
    if (isNetworkError(err)) {
      handleOfflineProfile();
      return;
    }
    bodyEl.innerHTML = `<div style="padding: 40px 20px; color: #ef4444; font-size: 14px;">Error loading profile: ${escapeHTML(
      err.message,
    )}</div>`;
  }
}
