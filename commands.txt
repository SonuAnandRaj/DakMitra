npx prettier --write . // to format whole document with prettier

# 1. HTML + Config + Edge Functions
Get-ChildItem -Recurse -Include *.html,*.json,*.ts | Where-Object { $_.FullName -notmatch '\\(node_modules|\.git)\\' } | ForEach-Object { "`n// ===== FILE: $($_.Name) =====`n" + (Get-Content $_.FullName -Raw) } | Set-Content bundle-1-html-config.txt -Encoding UTF8

# 2. All CSS
Get-ChildItem css -Include *.css -Recurse | ForEach-Object { "`n/* ===== FILE: $($_.Name) ===== */`n" + (Get-Content $_.FullName -Raw) } | Set-Content bundle-2-css.txt -Encoding UTF8

# 3. JS Core (01 to 05)
Get-ChildItem js -Recurse -Include 01-*.js,02-*.js,03-*.js,04-*.js,05-*.js | Sort-Object Name | ForEach-Object { "`n// ===== FILE: $($_.Name) =====`n" + (Get-Content $_.FullName -Raw) } | Set-Content bundle-3-js-core.txt -Encoding UTF8

# 4. JS Features (06 to 10)
Get-ChildItem js -Recurse -Include 06-*.js,07-*.js,08-*.js,09-*.js,10-*.js | Sort-Object Name | ForEach-Object { "`n// ===== FILE: $($_.Name) =====`n" + (Get-Content $_.FullName -Raw) } | Set-Content bundle-4-js-features.txt -Encoding UTF8
