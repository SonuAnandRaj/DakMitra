VS code to generate four small files

New-Item -ItemType Directory -Force -Path ai_upload | Out-Null

# 1. HTML + Config + Edge Functions
Get-ChildItem -File -Recurse -Include *.html,*.json,*.ts | Where-Object { $_.FullName -notmatch '\\(node_modules|\.git|ai_upload)\\' } | ForEach-Object { "`n// ===== FILE: $($_.Name) =====`n" + (Get-Content $_.FullName -Raw) } | Set-Content ai_upload\bundle-1-html-config.txt -Encoding UTF8

# 2. All CSS
Get-ChildItem css -File -Recurse -Include *.css | Sort-Object Name | ForEach-Object { "`n/* ===== FILE: $($_.Name) ===== */`n" + (Get-Content $_.FullName -Raw) } | Set-Content ai_upload\bundle-2-css.txt -Encoding UTF8

# 3. JS Core (01 to 05)
Get-ChildItem js -File -Recurse -Include 01-*.js,02-*.js,03-*.js,04-*.js,05-*.js | Sort-Object Name | ForEach-Object { "`n// ===== FILE: $($_.Name) =====`n" + (Get-Content $_.FullName -Raw) } | Set-Content ai_upload\bundle-3-js-core.txt -Encoding UTF8

# 4. JS Features (06 to 10)
Get-ChildItem js -File -Recurse -Include 06-*.js,07-*.js,08-*.js,09-*.js,10-*.js | Sort-Object Name | ForEach-Object { "`n// ===== FILE: $($_.Name) =====`n" + (Get-Content $_.FullName -Raw) } | Set-Content ai_upload\bundle-4-js-features.txt -Encoding UTF8
